import { useEffect, useState } from 'react';

export const DropboxDownload = ({ 
  url = "https://www.dropbox.com/scl/fi/t0jy6t4su47f92x8dznd5/out.zip?rlkey=6w5ro9zpfpr72p3vhhtbds5qf&st=zz00tqr8&dl=0", 
  fileName = "archive.zip",
  theme = "sunset"
}) => {
  const [downloading, setDownloading] = useState(false);

  const themes = {
    sunset: "from-amber-500 via-rose-500 to-purple-600 hover:shadow-rose-500/40",
    neon: "from-cyan-400 via-fuchsia-500 to-indigo-500 hover:shadow-fuchsia-500/40",
    emerald: "from-emerald-400 via-teal-500 to-cyan-600 hover:shadow-teal-500/40",
    ocean: "from-blue-500 via-indigo-500 to-cyan-400 hover:shadow-blue-500/40",
    purple: "from-purple-600 via-pink-500 to-red-500 hover:shadow-pink-500/40",
  };

  const activeTheme = themes[theme] || themes.sunset;

  useEffect(() => {
    if (!downloading) return;

    const directUrl = url.replace('dl=0', 'dl=1');

    fetch(directUrl)
      .then((res) => res.blob())
      .then((blob) => {
        const a = document.createElement('a');
        a.href = URL.createObjectURL(blob);
        a.download = fileName;
        a.click();
        URL.revokeObjectURL(a.href);
      })
      .catch(() => window.open(directUrl, '_blank'))
      .finally(() => setDownloading(false));
  }, [downloading, url, fileName]);

  return (
    <div className="my-8 flex justify-center">
      <button
        onClick={() => setDownloading(true)}
        disabled={downloading}
        className={`group relative inline-flex items-center justify-center gap-4 px-10 py-5 text-xl font-extrabold text-white rounded-2xl shadow-2xl transition-all duration-300 transform hover:scale-105 active:scale-95 ${
          downloading
            ? 'bg-slate-700 cursor-wait opacity-80'
            : `bg-linear-to-r ${activeTheme} bg-size-[200%_200%] animate-gradient hover:shadow-2xl`
        }`}
      >
        {/* Glow effect */}
        {!downloading && (
          <span className={`absolute -inset-1 rounded-2xl bg-linear-to-r ${activeTheme} opacity-70 blur-lg transition duration-500 group-hover:opacity-100 group-hover:blur-xl animate-gradient -z-10`} />
        )}

        {downloading ? (
          <>
            <svg className="w-8 h-8 animate-spin text-white" viewBox="0 0 24 24" fill="none">
              <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4" />
              <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z" />
            </svg>
            <span className="tracking-wide">Скачивание...</span>
          </>
        ) : (
          <>
            <svg className="w-8 h-8 transition-transform duration-300 group-hover:scale-110" fill="none" stroke="currentColor" viewBox="0 0 24 24">
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2.5" d="M4 16v1a3 3 0 003 3h10a3 3 0 003-3v-1m-4-4l-4 4m0 0l-4-4m4 4V4" />
            </svg>
            <span className="tracking-wide">Скачать {fileName}</span>
          </>
        )}
      </button>
    </div>
  );
};

export const FullPageBackground = () => {
  const [mouse, setMouse] = useState({ x: 0, y: 0 });

  useEffect(() => {
    const handleMouseMove = (e) => {
      const x = (e.clientX / window.innerWidth - 0.5) * 2;
      const y = (e.clientY / window.innerHeight - 0.5) * 2;
      setMouse({ x, y });
    };

    window.addEventListener('mousemove', handleMouseMove);
    return () => window.removeEventListener('mousemove', handleMouseMove);
  }, []);

  return (
    <div className="fixed inset-0 -z-10 w-full h-full pointer-events-none overflow-hidden bg-[hsl(73,90%,71%)]">
      <div 
        className="absolute top-[-20%] left-[-20%] w-[90vw] h-[90vw] rounded-full bg-[hsl(271,100%,50%)] blur-[140px] transition-transform duration-700 ease-out"
        style={{
          transform: `translate3d(${mouse.x * 60}px, ${mouse.y * 60}px, 0)`
        }}
      />
      
      <div 
        className="absolute bottom-[-20%] right-[-10%] w-[90vw] h-[90vw] rounded-full bg-[hsl(108,99%,29%)] blur-[140px] transition-transform duration-1000 ease-out"
        style={{
          transform: `translate3d(${mouse.x * -80}px, ${mouse.y * -80}px, 0)`
        }}
      />

      <div 
        className="absolute top-[20%] right-[-20%] w-[80vw] h-[80vw] rounded-full bg-[hsl(308,100%,50%)] blur-[140px] transition-transform duration-500 ease-out"
        style={{
          transform: `translate3d(${mouse.y * 50}px, ${mouse.x * -50}px, 0)`
        }}
      />
    </div>
  );
};

export const SoundButton = ({ emoji, soundText, bgColor }) => {
  const [active, setActive] = useState(false);
  const play = () => {
    setActive(true);
    setTimeout(() => setActive(false), 800);
  };
  return (
    <button 
      onClick={play}
      className={`${bgColor} text-white font-bold text-lg md:text-xl px-6 py-4 rounded-3xl shadow-lg hover:scale-105 active:scale-95 transition-transform cursor-pointer flex flex-col items-center justify-center gap-1 border-4 border-white`}
    >
      <span className="text-4xl">{emoji}</span>
      <span>{active ? soundText : "Нажми меня!"}</span>
    </button>
  );
};
